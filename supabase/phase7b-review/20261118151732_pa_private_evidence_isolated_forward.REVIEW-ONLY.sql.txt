CREATE TABLE public.pa_progress_evidence (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  employee_id uuid NOT NULL,
  individual_goal_id text NOT NULL,
  progress_update_id text NOT NULL
    REFERENCES public.pa_progress_updates(id) ON DELETE CASCADE,
  slot smallint NOT NULL CHECK (slot BETWEEN 1 AND 3),
  storage_path text NOT NULL UNIQUE,
  file_name text NOT NULL CHECK (char_length(btrim(file_name)) BETWEEN 1 AND 180),
  content_type text NOT NULL CHECK (content_type IN
    ('application/pdf','image/png','image/jpeg','image/webp')),
  size_bytes bigint NOT NULL CHECK (size_bytes BETWEEN 1 AND 10485760),
  uploaded_by uuid NOT NULL,
  status text NOT NULL DEFAULT 'pending' CHECK (status IN ('pending','ready')),
  created_at timestamptz NOT NULL DEFAULT now(),
  expires_at timestamptz NOT NULL DEFAULT (now() + interval '30 minutes'),
  finalized_at timestamptz,
  CONSTRAINT pa_evidence_unique_slot UNIQUE (progress_update_id,slot),
  CONSTRAINT pa_evidence_storage_path_shape CHECK (
    storage_path ~ '^[0-9a-f-]{36}/[0-9a-f-]{36}/[^/]+/[0-9a-f-]{36}[.](pdf|png|jpg|webp)$'
  ),
  CONSTRAINT pa_evidence_finalized_consistency CHECK (
    (status='pending' AND finalized_at IS NULL)
    OR (status='ready' AND finalized_at IS NOT NULL)
  )
);
CREATE INDEX pa_evidence_tenant_progress_idx

-- Reserve a slot before Storage upload. Caller-provided status/timing are ignored.
-- Triggers deliberately check protected parent with definer rights; no API privileges.
CREATE FUNCTION bexhr_private.bexhr_pa_evidence_reservation_guard()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$
DECLARE
  v_now timestamptz := now();
BEGIN
  IF auth.uid() IS NULL OR NEW.uploaded_by IS DISTINCT FROM auth.uid()
     OR NEW.tenant_id IS DISTINCT FROM public.current_authenticated_profile_tenant_id()
     OR NEW.employee_id IS DISTINCT FROM public.pa_current_employee_id() THEN
    RAISE EXCEPTION 'PA evidence reservation denied' USING ERRCODE='42501';
  END IF;
  IF NOT EXISTS (
    SELECT 1 FROM public.pa_progress_updates p
    WHERE p.id=NEW.progress_update_id AND p.tenant_id=NEW.tenant_id
      AND p.employee_id=NEW.employee_id
      AND p.individual_goal_id=NEW.individual_goal_id
    FOR KEY SHARE
  ) THEN RAISE EXCEPTION 'PA evidence parent mismatch' USING ERRCODE='23514'; END IF;
  IF split_part(NEW.storage_path,'/',1) IS DISTINCT FROM NEW.tenant_id::text
     OR split_part(NEW.storage_path,'/',2) IS DISTINCT FROM NEW.employee_id::text
     OR split_part(NEW.storage_path,'/',3) IS DISTINCT FROM NEW.progress_update_id THEN
    RAISE EXCEPTION 'PA evidence storage path mismatch' USING ERRCODE='23514';
  END IF;
  IF (NEW.content_type='application/pdf' AND right(NEW.storage_path,4)<>'.pdf')
     OR (NEW.content_type='image/png' AND right(NEW.storage_path,4)<>'.png')
     OR (NEW.content_type='image/jpeg' AND right(NEW.storage_path,4)<>'.jpg')
     OR (NEW.content_type='image/webp' AND right(NEW.storage_path,5)<>'.webp') THEN
    RAISE EXCEPTION 'PA evidence type/path mismatch' USING ERRCODE='23514';
  END IF;
  NEW.status := 'pending';
  NEW.created_at := v_now;
  NEW.expires_at := v_now + interval '30 minutes';
  NEW.finalized_at := NULL;
  RETURN NEW;
END;
$fn$;
REVOKE ALL ON FUNCTION bexhr_private.bexhr_pa_evidence_reservation_guard()
  FROM PUBLIC,anon,authenticated;
CREATE TRIGGER bexhr_pa_evidence_reservation_guard
BEFORE INSERT ON public.pa_progress_evidence
FOR EACH ROW EXECUTE FUNCTION bexhr_private.bexhr_pa_evidence_reservation_guard();

-- Do not permit metadata deletion while physical Storage object still exists.
-- Storage objects must be removed through the Storage API, never SQL.
CREATE FUNCTION bexhr_private.bexhr_pa_evidence_delete_guard()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$
BEGIN
  IF EXISTS (SELECT 1 FROM storage.objects o
             WHERE o.bucket_id='pa-progress-evidence' AND o.name=OLD.storage_path) THEN
    RAISE EXCEPTION 'PA evidence Storage object must be removed first'
      USING ERRCODE='23503';
  END IF;
  RETURN OLD;
END;
$fn$;
REVOKE ALL ON FUNCTION bexhr_private.bexhr_pa_evidence_delete_guard()
  FROM PUBLIC,anon,authenticated;
CREATE TRIGGER bexhr_pa_evidence_delete_guard
BEFORE DELETE ON public.pa_progress_evidence
FOR EACH ROW EXECUTE FUNCTION bexhr_private.bexhr_pa_evidence_delete_guard();

-- Only the owner can finalize, and only when an actual Storage object matches.
CREATE FUNCTION public.bexhr_pa_evidence_finalize(p_evidence_id uuid)
RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$
DECLARE
  v public.pa_progress_evidence%ROWTYPE;
  v_meta jsonb;
  v_owner text;
BEGIN
  IF auth.uid() IS NULL OR p_evidence_id IS NULL THEN
    RAISE EXCEPTION 'PA evidence finalize denied' USING ERRCODE='42501';
  END IF;
  SELECT * INTO v FROM public.pa_progress_evidence
  WHERE id=p_evidence_id FOR UPDATE;
  IF NOT FOUND OR v.status<>'pending' OR now()>v.expires_at
     OR v.uploaded_by IS DISTINCT FROM auth.uid()
     OR v.tenant_id IS DISTINCT FROM public.current_authenticated_profile_tenant_id()
     OR v.employee_id IS DISTINCT FROM public.pa_current_employee_id() THEN
    RAISE EXCEPTION 'PA evidence reservation expired or forbidden' USING ERRCODE='42501';
  END IF;
  SELECT o.metadata, o.owner_id INTO v_meta, v_owner FROM storage.objects o
  WHERE o.bucket_id='pa-progress-evidence' AND o.name=v.storage_path;
  IF NOT FOUND OR v_meta->>'mimetype' IS DISTINCT FROM v.content_type
     OR coalesce(v_meta->>'size','') !~ '^[0-9]+$'
     OR (v_meta->>'size')::bigint IS DISTINCT FROM v.size_bytes
     OR v_owner IS DISTINCT FROM auth.uid()::text THEN
    RAISE EXCEPTION 'PA evidence Storage MIME/size/owner verification failed'
      USING ERRCODE='23514';
  END IF;
  UPDATE public.pa_progress_evidence
    SET status='ready',finalized_at=now() WHERE id=p_evidence_id;
  RETURN TRUE;
END;
$fn$;
REVOKE ALL ON FUNCTION public.bexhr_pa_evidence_finalize(uuid) FROM PUBLIC,anon;
GRANT EXECUTE ON FUNCTION public.bexhr_pa_evidence_finalize(uuid) TO authenticated;

ON storage.objects FOR INSERT TO authenticated

  bucket_id='pa-progress-evidence' AND auth.uid() IS NOT NULL
  AND name ~ '^[0-9a-f-]{36}/[0-9a-f-]{36}/[^/]+/[0-9a-f-]{36}[.](pdf|png|jpg|webp)$'

    SELECT 1 FROM public.pa_progress_evidence e

ON storage.objects FOR SELECT TO authenticated

        (e.status='pending' AND e.uploaded_by=auth.uid())
